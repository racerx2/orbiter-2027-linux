# not upstream: collision and impact damage addon (Design CA); in-tree under Src/Plugin or standalone against an installed Orbiter
cmake_minimum_required(VERSION 3.21)
set(COLL_VERSION 0.1.0)

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(Collision VERSION ${COLL_VERSION} LANGUAGES CXX)
	set(COLL_STANDALONE ON)
	set(CMAKE_CXX_STANDARD 20)
	set(CMAKE_CXX_STANDARD_REQUIRED ON)
	set(CMAKE_CXX_EXTENSIONS OFF)
	set(ORBITER_DIR "" CACHE PATH "Installed Orbiter folder (the one holding Orbitersdk/)")
	if (NOT EXISTS "${ORBITER_DIR}/Orbitersdk/include/OrbiterAPI.h")
		message(FATAL_ERROR "ORBITER_DIR must name an installed Orbiter with Orbitersdk/include")
	endif ()
	set(COLL_SDK_INCLUDE ${ORBITER_DIR}/Orbitersdk/include)
	option(COLLISION_TESTS "Build the addon tests" ON)
	set(COLLISION_TEST_DATA "" CACHE PATH "An Orbiter source tree with the stock meshes some unit tests read")
	set(COLL_MAKE_TESTS ${COLLISION_TESTS})
	set(COLL_TEST_DATA ${COLLISION_TEST_DATA})
	set(COLL_TEST_BIN ${CMAKE_CURRENT_BINARY_DIR}/Tests)
	set(COLL_TEST_WD ${CMAKE_CURRENT_BINARY_DIR})
else ()
	set(COLL_STANDALONE OFF)
	set(COLL_SDK_INCLUDE ${ORBITER_SOURCE_SDK_INCLUDE_DIR})
	set(COLL_MAKE_TESTS ${ORBITER_MAKE_TESTS})
	set(COLL_TEST_DATA ${ORBITER_SOURCE_ROOT_DIR})
	set(COLL_TEST_BIN ${ORBITER_BINARY_ROOT_DIR}/Tests)
	set(COLL_TEST_WD ${ORBITER_BINARY_ROOT_DIR})
endif ()

set(CMAKE_AUTOMOC OFF)
set_property(DIRECTORY PROPERTY LINK_LIBRARIES "") # drops Src/Plugin's Qt6::Widgets
if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
	add_compile_options(-fno-gnu-unique) # every target here: STB_GNU_UNIQUE keeps the module mapped after dlclose, so a re-tick would skip InitModule
endif ()
if (MSVC)
	add_compile_definitions(NOMINMAX)
endif ()

set(COLL_UNITS Vecmat CollGeom CollAnim CollShape CollDetect CollSolve CollSolveFrame DentMath)
foreach (u ${COLL_UNITS})
	add_library(Coll_${u} OBJECT ${u}.cpp)
	target_include_directories(Coll_${u} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${COLL_SDK_INCLUDE})
	set_target_properties(Coll_${u} PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON FOLDER Modules/Collision)
endforeach ()
if (CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
	target_compile_options(Coll_DentMath PRIVATE -ffp-contract=off) # dent geometry must not depend on FMA contraction
endif ()

set(COLL_TESTFILE ${CMAKE_CURRENT_BINARY_DIR}/CollisionTests.cmake)
file(WRITE ${COLL_TESTFILE} "# generated by Src/Plugin/Collision/CMakeLists.txt\n")
function(coll_add_test name)
	cmake_parse_arguments(PARSE_ARGV 1 T "" "TIMEOUT" "COMMAND;LABELS")
	set(cmd "")
	foreach (a ${T_COMMAND})
		string(APPEND cmd " [==[${a}]==]")
	endforeach ()
	file(APPEND ${COLL_TESTFILE} "add_test(NAME [==[${name}]==] COMMAND${cmd} WORKING_DIRECTORY [==[${COLL_TEST_WD}]==])\n")
	file(APPEND ${COLL_TESTFILE} "set_tests_properties([==[${name}]==] PROPERTIES TIMEOUT ${T_TIMEOUT} LABELS [==[${T_LABELS}]==])\n")
endfunction()

# the gate: no test target or registration exists without ORBITER_MAKE_TESTS (in-tree) or COLLISION_TESTS (standalone)
if (COLL_MAKE_TESTS)
	if (COLL_STANDALONE)
		find_package(Catch2 3 CONFIG QUIET)
		if (NOT Catch2_FOUND)
			include(FetchContent)
			FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.8.1) # as Tests/CMakeLists.txt; FETCHCONTENT_SOURCE_DIR_CATCH2 reuses a local copy
			set(BUILD_SHARED_LIBS OFF)
			FetchContent_MakeAvailable(Catch2)
		endif ()
	endif ()
	function(coll_unit_test name timeout)
		cmake_parse_arguments(PARSE_ARGV 2 U "DATA" "" "UNITS")
		if (U_DATA AND NOT COLL_TEST_DATA)
			return () # reads stock meshes: needs COLLISION_TEST_DATA standalone
		endif ()
		add_executable(${name} Tests/${name}.cpp)
		foreach (u ${U_UNITS})
			target_sources(${name} PRIVATE $<TARGET_OBJECTS:Coll_${u}>)
		endforeach ()
		target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/Tests ${COLL_SDK_INCLUDE})
		target_compile_definitions(${name} PRIVATE COLL_TEST_SRC="${COLL_TEST_DATA}" COLL_ADDON_SRC="${CMAKE_CURRENT_SOURCE_DIR}")
		target_link_libraries(${name} Catch2::Catch2WithMain)
		set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${COLL_TEST_BIN} FOLDER Tests/Collision)
		coll_add_test(${name} TIMEOUT ${timeout} COMMAND $<TARGET_FILE:${name}> LABELS collision)
	endfunction()
	function(coll_slow_test name suffix tag)
		if (TARGET ${name})
			coll_add_test(${name}.${suffix} TIMEOUT 600 COMMAND $<TARGET_FILE:${name}> [${tag}] LABELS collision slow)
		endif ()
	endfunction()
	coll_unit_test(CollGeom.Test 30 DATA UNITS CollGeom Vecmat)
	coll_unit_test(CollAnim.Test 30 DATA UNITS CollAnim Vecmat)
	coll_unit_test(CollShape.Test 30 DATA UNITS CollShape CollAnim CollGeom Vecmat)
	coll_unit_test(CollDetect.Test 120 DATA UNITS CollDetect CollGeom Vecmat)
	coll_unit_test(CollSolve.Test 30 UNITS CollSolve Vecmat)
	coll_unit_test(CollSolveFrame.Test 120 UNITS CollSolveFrame CollSolve CollDetect CollGeom Vecmat)
	coll_unit_test(DentMath.Test 30 DATA UNITS DentMath Vecmat)
	coll_unit_test(CollLoop.Test 120 DATA UNITS CollSolveFrame CollSolve CollDetect CollGeom DentMath Vecmat)
	coll_slow_test(CollGeom.Test Slow slow)
	coll_slow_test(CollAnim.Test Slow slow)
	coll_slow_test(CollShape.Test Slow slow)
	coll_slow_test(CollDetect.Test Fuzz fuzz)
	coll_slow_test(DentMath.Test Fuzz fuzz)
endif ()

if (COLL_STANDALONE)
	enable_testing()
	include(${COLL_TESTFILE})
else ()
	# deferred to the end of the root file, after its enable_testing(); EVAL expands the path now
	cmake_language(EVAL CODE "cmake_language(DEFER DIRECTORY [[${CMAKE_SOURCE_DIR}]] CALL include [[${COLL_TESTFILE}]])")
endif ()
