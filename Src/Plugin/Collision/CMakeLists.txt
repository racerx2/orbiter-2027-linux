function(coll_add_test name)
	cmake_parse_arguments(PARSE_ARGV 1 T "SERIAL" "TIMEOUT;SKIP" "COMMAND;LABELS;PROPS")
	set(cmd "")
	foreach (a ${T_COMMAND})
		string(APPEND cmd " [==[${a}]==]")
	endforeach ()
	file(APPEND ${COLL_TESTFILE} "add_test(NAME [==[${name}]==] COMMAND${cmd} WORKING_DIRECTORY [==[${COLL_TEST_WD}]==])\n")
	set(props "TIMEOUT ${T_TIMEOUT} LABELS [==[${T_LABELS}]==]")
	if (T_SKIP)
		string(APPEND props " SKIP_RETURN_CODE ${T_SKIP}")
	endif ()
	if (T_SERIAL)
		string(APPEND props " RUN_SERIAL TRUE")
	endif ()
	foreach (p ${T_PROPS})
		string(APPEND props " [==[${p}]==]")
	endforeach ()
	file(APPEND ${COLL_TESTFILE} "set_tests_properties([==[${name}]==] PROPERTIES ${props})\n")
endfunction()
