# GltfExport: a vessel with its animations as one .glb (claude/design-gltf-export.md); standalone against an Orbiter build or install
cmake_minimum_required(VERSION 3.21)
project(GltfExport VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if (CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
	add_compile_options(-fno-gnu-unique) # a re-ticked module must run InitModule again
endif ()

option(GLTFX_PLUGIN "Build the plugin module (needs ORBITER_DIR)" ON)
option(GLTFX_TESTS "Build the tests" ON)
set(ORBITER_DIR "" CACHE PATH "Orbiter folder holding Orbitersdk/ (a build tree or an install)")
get_filename_component(gltfx_src_default ${CMAKE_CURRENT_LIST_DIR}/../.. ABSOLUTE)
set(GLTFX_SOURCE_ROOT ${gltfx_src_default} CACHE PATH "The Orbiter source tree around this add-on (cmake/, Src/Orbiter/GltfLoader.cpp)")
set(GLTFX_SDK_INCLUDE "" CACHE PATH "Orbitersdk/include, when ORBITER_DIR is not given")
if (NOT GLTFX_SDK_INCLUDE)
	if (ORBITER_DIR)
		set(GLTFX_SDK_INCLUDE ${ORBITER_DIR}/Orbitersdk/include)
	else ()
		set(GLTFX_SDK_INCLUDE ${GLTFX_SOURCE_ROOT}/Orbitersdk/include)
	endif ()
endif ()

find_package(Qt6 REQUIRED COMPONENTS Core Gui)
set(CMAKE_AUTOMOC OFF)

add_library(GltfExportCore STATIC AnimModel.cpp DdsDecode.cpp MshText.cpp GlbWriter.cpp Exporter.cpp)
set_target_properties(GltfExportCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(GltfExportCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(GltfExportCore PUBLIC Qt6::Core Qt6::Gui)

if (GLTFX_PLUGIN)
	if (NOT EXISTS "${ORBITER_DIR}/Orbitersdk/include/OrbiterAPI.h")
		message(FATAL_ERROR "ORBITER_DIR must name an Orbiter folder with Orbitersdk/include")
	endif ()
	find_library(GLTFX_SDK_LIB Orbitersdk PATHS ${ORBITER_DIR}/Orbitersdk/lib NO_DEFAULT_PATH REQUIRED)
	set(gltfx_rc "")
	foreach (d ${ORBITER_DIR}/cmake ${GLTFX_SOURCE_ROOT}/cmake)
		if (NOT gltfx_rc AND EXISTS "${d}/rcresources.cmake" AND EXISTS "${d}/rc2cpp.py")
			set(gltfx_rc ${d})
		endif ()
	endforeach ()
	if (NOT gltfx_rc)
		message(FATAL_ERROR "GltfExport: rcresources.cmake and rc2cpp.py not found in ${ORBITER_DIR}/cmake or ${GLTFX_SOURCE_ROOT}/cmake")
	endif ()
	set(ORBITER_SOURCE_SDK_INCLUDE_DIR ${GLTFX_SDK_INCLUDE}) # rcresources.cmake searches it for the .rc includes
	include(${gltfx_rc}/rcresources.cmake)

	add_library(GltfExport SHARED GltfExport.cpp Snapshot.cpp)
	set_target_properties(GltfExport PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/Modules/Plugin)
	target_include_directories(GltfExport PRIVATE ${GLTFX_SDK_INCLUDE})
	target_link_libraries(GltfExport PRIVATE GltfExportCore ${GLTFX_SDK_LIB})
	target_link_options(GltfExport PRIVATE "LINKER:--undefined=GetModuleVersion") # pulls Orbitersdk.o, the module's ELF constructor that runs InitModule
	orbiter_rc_resources(GltfExport GltfExport.rc)
	if (CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
		set(CMAKE_INSTALL_PREFIX ${ORBITER_DIR} CACHE PATH "" FORCE)
	endif ()
	install(TARGETS GltfExport LIBRARY DESTINATION Modules/Plugin)
endif ()

if (GLTFX_TESTS)
	enable_testing()
	find_package(Catch2 3 CONFIG QUIET)
	if (NOT Catch2_FOUND)
		include(FetchContent)
		FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.8.1) # FETCHCONTENT_SOURCE_DIR_CATCH2 reuses a local copy
		set(BUILD_SHARED_LIBS OFF)
		FetchContent_MakeAvailable(Catch2)
	endif ()
	function(gltfx_test name)
		add_executable(${name} Tests/${name}.cpp ${ARGN})
		target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Tests ${GLTFX_SDK_INCLUDE} ${GLTFX_SOURCE_ROOT}/Src/Orbiter)
		target_compile_definitions(${name} PRIVATE GLTFX_SOURCE_ROOT="${GLTFX_SOURCE_ROOT}")
		target_link_libraries(${name} PRIVATE GltfExportCore Catch2::Catch2WithMain)
		add_test(NAME ${name} COMMAND ${name})
	endfunction()
	gltfx_test(AnimModel.Test)
	gltfx_test(DdsDecode.Test)
	gltfx_test(MshText.Test)
	gltfx_test(Glb.Test ${GLTFX_SOURCE_ROOT}/Src/Orbiter/GltfLoader.cpp ${GLTFX_SOURCE_ROOT}/Src/Orbiter/GltfRoundTrip.cpp)
	gltfx_test(RoundTrip.Test ${GLTFX_SOURCE_ROOT}/Src/Orbiter/GltfLoader.cpp ${GLTFX_SOURCE_ROOT}/Src/Orbiter/GltfRoundTrip.cpp)
	add_executable(gltfx_ref Tests/gltfx_ref.cpp)
	target_include_directories(gltfx_ref PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/Tests)
	target_link_libraries(gltfx_ref PRIVATE GltfExportCore)
endif ()
